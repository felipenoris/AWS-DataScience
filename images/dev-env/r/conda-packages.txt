# The R environment, as conda-forge specs - one per line, comments and blanks ignored.
#
# WHY CONDA AND NOT apt OR CRAN DIRECTLY: CodeArtifact has no CRAN external connection
# (docs/plan/architecture.md §4.3), so under design B R arrives baked into this image. That
# makes reproducibility the whole point, and conda-forge is the one channel that pins the R
# interpreter and its compiled packages together. Everything below is a prebuilt binary - no
# compilation, which is what keeps the amd64-under-emulation build on an Apple Silicon laptop
# merely slow rather than impossible.
#
# IT IS ITS OWN ENVIRONMENT (/opt/conda/envs/r), never the base one. Solving R into the
# SageMaker Distribution's base environment would let the solver move Python packages the
# distribution pinned on purpose.
#
# PIN THE INTERPRETER, let the solver pick compatible builds for the rest.

r-irkernel=1.3.2
